cmake_minimum_required(VERSION 3.20)
project(Scalars VERSION 0.1.0 LANGUAGES CXX)
include(GNUInstallDirs)
include(CMakePackageConfigHelpers)
option(SCALARS_BUILD_TESTS "Build scalar tests" ON)
add_library(ScalarsCore INTERFACE)
add_library(Scalars::Core ALIAS ScalarsCore)
target_compile_features(ScalarsCore INTERFACE cxx_std_20)
target_include_directories(ScalarsCore INTERFACE $<BUILD_INTERFACE:${CMAKE_CURRENT_SOURCE_DIR}/include> $<INSTALL_INTERFACE:${CMAKE_INSTALL_INCLUDEDIR}>)
add_library(Scalars STATIC src/core/ScalarType.cpp)
add_library(Scalars::Scalars ALIAS Scalars)
target_link_libraries(Scalars PUBLIC ScalarsCore)
set_target_properties(Scalars PROPERTIES EXPORT_NAME Scalars)
set_target_properties(ScalarsCore PROPERTIES EXPORT_NAME Core)
install(DIRECTORY include/ DESTINATION ${CMAKE_INSTALL_INCLUDEDIR})
install(TARGETS Scalars ScalarsCore EXPORT ScalarsTargets ARCHIVE DESTINATION ${CMAKE_INSTALL_LIBDIR})
install(EXPORT ScalarsTargets NAMESPACE Scalars:: DESTINATION ${CMAKE_INSTALL_LIBDIR}/cmake/Scalars)
configure_package_config_file(cmake/ScalarsConfig.cmake.in ${CMAKE_CURRENT_BINARY_DIR}/ScalarsConfig.cmake INSTALL_DESTINATION ${CMAKE_INSTALL_LIBDIR}/cmake/Scalars)
write_basic_package_version_file(${CMAKE_CURRENT_BINARY_DIR}/ScalarsConfigVersion.cmake VERSION ${PROJECT_VERSION} COMPATIBILITY SameMajorVersion)
install(FILES ${CMAKE_CURRENT_BINARY_DIR}/ScalarsConfig.cmake ${CMAKE_CURRENT_BINARY_DIR}/ScalarsConfigVersion.cmake DESTINATION ${CMAKE_INSTALL_LIBDIR}/cmake/Scalars)
if(SCALARS_BUILD_TESTS)
 enable_testing()
 add_executable(ScalarsTests tests/TestScalars.cpp)
 target_link_libraries(ScalarsTests PRIVATE Scalars::Scalars)
 add_test(NAME ScalarsTests COMMAND ScalarsTests)
endif()
